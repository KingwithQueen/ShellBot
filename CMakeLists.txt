cmake_minimum_required(VERSION 3.21)
project(ShellBot VERSION 1.0.3 LANGUAGES C)
if(WIN32)
  message(FATAL_ERROR "Windows uses scripts/build.ps1 and the existing MSBuild projects.")
endif()
if(NOT APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This backend currently targets Linux and macOS.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SHELLBOT_TESTS "Enable the local-only POSIX integration suite" OFF)
option(SHELLBOT_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(SHELLBOT_DESKTOP "Build the native management window and tray" ON)
find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
add_library(sb_json STATIC third_party/yyjson/yyjson.c)
target_include_directories(sb_json PUBLIC third_party/yyjson)
add_library(sb_http STATIC third_party/civetweb/src/civetweb.c)
target_include_directories(sb_http PUBLIC third_party/civetweb/include)
target_compile_definitions(sb_http PRIVATE NO_SSL NO_CGI NO_FILES NO_FILESYSTEM NO_WEBSOCKET USE_IPV6=0)
target_link_libraries(sb_http PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
add_executable(ShellBot
  PosixShellBot/main.c PosixShellBot/transport.c PosixShellBot/owner.c
  PosixShellBot/platform.c PosixShellBot/sha256.c PosixShellBot/unavailable.c
  PosixShellBot/process.c PosixShellBot/execution.c PosixShellBot/task_agent.c
  PosixShellBot/manager.c PosixShellBot/credentials.c PosixShellBot/keyring.c PosixShellBot/oauth.c
  PosixShellBot/desktop.c PosixShellBot/desktop_png.c
  core/config/config.c core/config/projects.c core/protocol/mcp.c
  core/policy/approval.c core/policy/audit.c core/policy/state.c core/policy/project_policy.c
  core/tools/tools.c core/tools/text.c core/tools/search.c core/tools/tasks.c
  core/tools/image.c core/tools/png_validate.c core/tools/document.c
  core/tools/transfer.c core/tools/desktop.c)
target_include_directories(ShellBot PRIVATE core/include PosixShellBot)
target_compile_definitions(ShellBot PRIVATE SB_POSIX=1 _FILE_OFFSET_BITS=64)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(ShellBot PRIVATE _GNU_SOURCE)
else()
  target_compile_definitions(ShellBot PRIVATE _DARWIN_C_SOURCE)
endif()
target_compile_options(ShellBot PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
target_link_libraries(ShellBot PRIVATE sb_json sb_http Threads::Threads ZLIB::ZLIB)
if(APPLE)
  enable_language(OBJC)
  target_sources(ShellBot PRIVATE PosixShellBot/desktop_macos.m)
  set_source_files_properties(PosixShellBot/desktop_macos.m PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(ShellBot PRIVATE "-framework Cocoa" "-framework ApplicationServices"
    "-framework ScreenCaptureKit" "-framework CoreMedia" "-framework CoreVideo")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DESKTOP_NATIVE REQUIRED IMPORTED_TARGET x11 xtst xi xfixes xrandr gio-unix-2.0 gstreamer-app-1.0 gstreamer-video-1.0)
  target_sources(ShellBot PRIVATE PosixShellBot/desktop_linux.c)
  target_link_libraries(ShellBot PRIVATE PkgConfig::DESKTOP_NATIVE)
endif()
if(SHELLBOT_DESKTOP)
  target_sources(ShellBot PRIVATE PosixShellBot/native.c)
  target_compile_definitions(ShellBot PRIVATE SB_NATIVE_DESKTOP=1)
  if(APPLE)
    enable_language(OBJC)
    target_sources(ShellBot PRIVATE PosixShellBot/cocoa.m)
    set_source_files_properties(PosixShellBot/cocoa.m PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(ShellBot PRIVATE "-framework Cocoa")
    add_custom_command(TARGET ShellBot POST_BUILD
      COMMAND ${CMAKE_COMMAND} "-DOUTPUT=${CMAKE_BINARY_DIR}" "-DSOURCE=${CMAKE_SOURCE_DIR}" "-DPROGRAM=$<TARGET_FILE:ShellBot>" -P "${CMAKE_SOURCE_DIR}/PosixShellBot/bundle.cmake")
    install(DIRECTORY "${CMAKE_BINARY_DIR}/ShellBot.app" DESTINATION . USE_SOURCE_PERMISSIONS)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.10)
    target_sources(ShellBot PRIVATE PosixShellBot/gtk.c)
    target_link_libraries(ShellBot PRIVATE PkgConfig::GTK4)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/share/ShellBot")
    configure_file(assets/branding/shellbot-v1/shellbot-icon-small.png share/ShellBot/shellbot-icon-small.png COPYONLY)
    install(FILES PosixShellBot/org.shellbot.desktop.desktop DESTINATION share/applications)
    install(FILES assets/branding/shellbot-v1/shellbot-icon.png DESTINATION share/icons/hicolor/512x512/apps RENAME shellbot.png)
    install(FILES assets/branding/shellbot-v1/shellbot-icon-small.png DESTINATION share/ShellBot)
  endif()
endif()
if(APPLE)
  target_link_libraries(ShellBot PRIVATE "-framework Security" "-framework CoreFoundation")
else()
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBSECRET REQUIRED IMPORTED_TARGET libsecret-1)
  target_link_libraries(ShellBot PRIVATE OpenSSL::Crypto PkgConfig::LIBSECRET)
endif()
if(SHELLBOT_SANITIZERS)
  target_compile_options(ShellBot PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ShellBot PRIVATE -fsanitize=address,undefined)
endif()
if(SHELLBOT_TESTS)
  target_compile_definitions(ShellBot PRIVATE SHELLBOT_TESTS=1)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_library(sb_test_clock SHARED tests/posix_clock.c)
  target_link_libraries(sb_test_clock PRIVATE ${CMAKE_DL_LIBS})
  foreach(test_file posix_integration.py posix_phase2.py posix_oauth.py posix_clock.c posix_remote.py posix_remote_fixture.c)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test_file}")
      message(FATAL_ERROR "Restore local tests/${test_file} before enabling SHELLBOT_TESTS.")
    endif()
  endforeach()
  add_test(NAME posix-integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/posix_integration.py $<TARGET_FILE:ShellBot>)
  set_tests_properties(posix-integration PROPERTIES TIMEOUT 240)
  add_test(NAME remote-desktop COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/posix_remote.py $<TARGET_FILE:ShellBot>)
  set_tests_properties(remote-desktop PROPERTIES TIMEOUT 150)
  if(SHELLBOT_DESKTOP)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/posix_desktop.py")
      message(FATAL_ERROR "Restore local tests/posix_desktop.py before enabling desktop tests.")
    endif()
    add_test(NAME desktop-controller COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/posix_desktop.py $<TARGET_FILE:ShellBot>)
    set_tests_properties(desktop-controller PROPERTIES TIMEOUT 180)
  endif()
endif()
install(TARGETS ShellBot RUNTIME DESTINATION bin)
install(FILES LICENSE.txt DESTINATION share/ShellBot/licenses)
install(FILES third_party/yyjson/LICENSE DESTINATION share/ShellBot/licenses RENAME yyjson-LICENSE)
install(FILES third_party/civetweb/LICENSE.md DESTINATION share/ShellBot/licenses RENAME civetweb-LICENSE.md)
install(FILES docs/POSIX_BACKEND.md DESTINATION share/ShellBot)
install(FILES docs/POSIX_OAUTH.md DESTINATION share/ShellBot)
install(FILES docs/POSIX_DESKTOP.md DESTINATION share/ShellBot)
install(FILES docs/POSIX_REMOTE_DESKTOP.md DESTINATION share/ShellBot)
install(FILES config/posix-desktop.example.json DESTINATION share/ShellBot/examples)
install(FILES config/posix-projects.example.json DESTINATION share/ShellBot/examples)
install(FILES config/posix-commands.example.json DESTINATION share/ShellBot/examples)
install(FILES config/posix-oauth.example.json DESTINATION share/ShellBot/examples)
